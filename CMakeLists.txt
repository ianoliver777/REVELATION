cmake_minimum_required(VERSION 3.22)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS version")
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
project(Revelation VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(JUCE_DIR)
    add_subdirectory("${JUCE_DIR}" JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.12 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
set(formats VST3)
if(APPLE)
    list(APPEND formats AU)
endif()
juce_add_binary_data(RevelationArt SOURCES Assets/mark.png Assets/panel.png)
juce_add_plugin(Revelation COMPANY_NAME "TRIPLE SEVEN AUDIO"
    BUNDLE_ID world.triple7.revelation PLUGIN_MANUFACTURER_CODE T7Au PLUGIN_CODE Rv03
    PRODUCT_NAME "REVELATION" DESCRIPTION "REVELATION Mastering Finisher"
    VST3_CATEGORIES Fx AU_MAIN_TYPE kAudioUnitType_Effect
    COPY_PLUGIN_AFTER_BUILD FALSE FORMATS ${formats})
target_sources(Revelation PRIVATE Source/Plugin.cpp)
target_compile_definitions(Revelation PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Revelation PRIVATE RevelationArt juce::juce_audio_utils
    PUBLIC juce::juce_recommended_config_flags)
option(REVELATION_TESTS "Build the DSP and host smoke tests" OFF)
if(REVELATION_TESTS)
    enable_testing()
    add_executable(DspTest Tests/DspTest.cpp)
    target_include_directories(DspTest PRIVATE Source)
    add_test(NAME DSP COMMAND DspTest)
    add_executable(EditorPreview Tests/EditorPreview.cpp)
    target_include_directories(EditorPreview PRIVATE
        $<TARGET_PROPERTY:juce::juce_audio_utils,INTERFACE_INCLUDE_DIRECTORIES>)
    target_compile_definitions(EditorPreview PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
    target_link_libraries(EditorPreview PRIVATE Revelation)
    juce_add_console_app(HostTest PRODUCT_NAME "Revelation Host Test")
    target_sources(HostTest PRIVATE Tests/HostTest.cpp)
    target_compile_definitions(HostTest PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1
        JUCE_PLUGINHOST_AU=$<BOOL:${APPLE}>)
    target_link_libraries(HostTest PRIVATE juce::juce_audio_utils)
endif()
